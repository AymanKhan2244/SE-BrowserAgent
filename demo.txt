def run_iterative_project_builder(cleaned_spec: str) -> List[GeneratedFile]:
    # Phase 1: File Manifest Generation
    architect_chain = architect_prompt | llm 
    manifest = architect_chain.invoke({"cleaned_spec": cleaned_spec})
    
    print(f"📋 Project: {manifest.project_name}")
    print(f"📋 Plan: Generating {len(manifest.file_paths)} files independently...\n")
    
    generated_project_files: List[GeneratedFile] = []
    engineer_chain = engineer_prompt | structured_engineer_llm
    
    # Phase 2: Sequential File Generation Loop
    for file_path in manifest.file_paths:
        try:
            structured_file: GeneratedFile = engineer_chain.invoke({
                "cleaned_spec": cleaned_spec,
                "target_file_path": file_path
            })
           
            structured_file.path = file_path 
            #save_to_disk(structured_file)
            generated_project_files.append(structured_file)
            
        except Exception as e:
            print(f"❌ Failed to generate {file_path}: {str(e)}")
            
    return generated_project_files




##  file  saver function 


def save_to_disk(generated_file: GeneratedFile):
    """
    Saves a GeneratedFile inside the generated_project folder
    while preserving the relative folder structure.
    """

    relative_path = generated_file.path
    content = generated_file.content

    # Create full path inside generated_project
    full_path = os.path.join(BASE_DIR, relative_path)

    # Create parent directories if they don't exist
    os.makedirs(os.path.dirname(full_path), exist_ok=True)

    # Write file
    with open(full_path, "w", encoding="utf-8") as f:
        f.write(content)

    print(
        f"✅ Saved: {full_path} "
        f"({len(content.splitlines())} lines)"
    )

from __future__ import annotations

import os
import re
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional

from pydantic import BaseModel, Field
from langchain_core.prompts import ChatPromptTemplate


# ---------------------------------------------------------------------------
# Data models
# ---------------------------------------------------------------------------

class GeneratedFile(BaseModel):
    path: str = Field(description="Relative file path, e.g. 'backend/app/main.py'")
    content: str = Field(description="Complete, un-truncated source code for this file.")


class GeneratedProject(BaseModel):
    project_name: str
    files: List[GeneratedFile]


@dataclass
class BuildResult:
    project_name: str
    output_dir: Optional[str]
    files: List[GeneratedFile]
    ok: bool
    validation_errors: List[str] = field(default_factory=list)


# ---------------------------------------------------------------------------
# File ordering — generate dependency-relevant files first so later files
# can see real code (config/models/db) instead of guessing at names.
# ---------------------------------------------------------------------------

_PRIORITY_HINTS = [
    "requirements.txt",
    "package.json",
    "config.py",
    "settings.py",
    "database.py",
    "models.py",
    "schemas.py",
    "routes.py",
    "app.py",
    "main.py",
]


def _order_files(file_paths: List[str]) -> List[str]:
    def sort_key(path: str):
        basename = path.split("/")[-1]
        if basename in _PRIORITY_HINTS:
            return (0, _PRIORITY_HINTS.index(basename))
        if basename.endswith(".py"):
            return (1, path)
        if "templates/" in path or basename.endswith(".html"):
            return (2, path)
        if "static/" in path or basename.endswith((".css", ".js")):
            return (3, path)
        return (4, path)

    return sorted(file_paths, key=sort_key)


def _strip_code_fences(text: str) -> str:
    """Safety net in case the model wraps content in ```lang fences
    despite structured output."""
    text = text.strip()
    text = re.sub(r"^```[a-zA-Z]*\n?", "", text)
    text = re.sub(r"\n?```$", "", text)
    return text.strip()


def _safe_invoke(chain, inputs: dict, retries: int = 3, backoff: float = 1.5):
    """Retry wrapper for structured-output calls that can fail with
    transient parse errors or truncated tool-call JSON."""
    last_err = None
    for attempt in range(retries):
        try:
            return chain.invoke(inputs)
        except Exception as e:  # noqa: BLE001 - want to inspect message text
            last_err = e
            msg = str(e)
            transient = (
                "parse tool call arguments" in msg
                or "invalid_request_error" in msg
                or "tool_use_failed" in msg
                or "failed_generation" in msg
            )
            if transient:
                time.sleep(backoff * (attempt + 1))
                continue
            raise
    raise RuntimeError(f"Structured output failed after {retries} attempts: {last_err}")

ENGINEER_PROMPT = ChatPromptTemplate.from_template(
    """You are a principal software engineer implementing ONE source file
in a multi-file project.

PROJECT SPEC / ARCHITECTURE:
{architecture_notes}

ALREADY GENERATED FILES — use these exactly as-is. Match their function
names, class names, imports, route paths, and variable names exactly.
Do not invent alternate names for things that already exist below.
{already_generated_files}

TARGET FILE:
{target_file_path}

Write the complete implementation for this file only. Include all imports,
full class/function bodies, and correct references to the files above.
Never generate placeholders, TODOs, or pseudocode.

Return only the source code for {target_file_path} in the `content` field."""
)


class CodingAgent:
    """Generates a full project from an architect's file manifest."""

    def __init__(self, llm, max_tokens: int = 8192):
        if isinstance(llm, (str, os.PathLike)) or not hasattr(
            llm, "with_structured_output"
        ):
            raise TypeError(
                "CodingAgent(llm) expects a LangChain chat model (e.g. "
                f"ChatGroq(...)), got {type(llm).__name__!r}. If you meant "
                "to pass an output directory, that goes into build(..., "
                "output_dir=...) instead, e.g.:\n"
                "  agent = CodingAgent(coding_llm)\n"
                "  agent.build(architect_output, architecture_notes=..., "
                "output_dir='generated_project')"
            )
        self.llm = llm
        self.structured_llm = llm.bind(max_tokens=max_tokens).with_structured_output(
            GeneratedFile
        )
        self.engineer_chain = ENGINEER_PROMPT | self.structured_llm

    def build(
        self,
        architect_output,
        architecture_notes: str,
        output_dir: Optional[str] = None,
    ) -> BuildResult:
        """Generate every file described by `architect_output`, in
        dependency order, each file seeing all previously generated files
        as context.

        architect_output: your ProjectManifest object — must expose
            `.project_name` (str) and `.file_paths` (list[str]).
        architecture_notes: the spec text to ground the implementation,
            e.g. str(cleaned_spec) from your planner_agent().
        """
        if output_dir is not None and not isinstance(output_dir, (str, os.PathLike)):
            raise TypeError(
                "output_dir must be a str or path-like, got "
                f"{type(output_dir).__name__!r}. This usually means an "
                "argument landed in the wrong position — call build() with "
                "output_dir='...' as a keyword, e.g.:\n"
                "  agent.build(architect_output, architecture_notes=..., "
                "output_dir='generated_project')"
            )
        if not hasattr(architect_output, "project_name") or not hasattr(
            architect_output, "file_paths"
        ):
            raise TypeError(
                "architect_output must be your ProjectManifest object "
                "(with .project_name and .file_paths), got "
                f"{type(architect_output).__name__!r}."
            )

        project_name = architect_output.project_name
        file_paths = architect_output.file_paths
        ordered_paths = _order_files(file_paths)
        generated: Dict[str, str] = {}
        errors: List[str] = []

        print(f"Project: {project_name}")
        print(f"Generating {len(ordered_paths)} files in dependency order...\n")

        for file_path in ordered_paths:
            context = (
                "\n\n".join(f"# --- {p} ---\n{c}" for p, c in generated.items())
                or "(none yet — this is the first file)"
            )
            try:
                result = _safe_invoke(
                    self.engineer_chain,
                    {
                        "architecture_notes": architecture_notes,
                        "already_generated_files": context,
                        "target_file_path": file_path,
                    },
                )
                generated[file_path] = _strip_code_fences(result.content)
                print(f"  ok  {file_path}")
            except Exception as e:  # noqa: BLE001
                msg = f"Failed to generate {file_path}: {e}"
                print(f"  FAIL {file_path}: {e}")
                errors.append(msg)
                generated[file_path] = f"# GENERATION FAILED: {e}"

        files = [GeneratedFile(path=p, content=c) for p, c in generated.items()]

        if output_dir:
            self._write_to_disk(files, output_dir)

        return BuildResult(
            project_name=project_name,
            output_dir=output_dir,
            files=files,
            ok=len(errors) == 0,
            validation_errors=errors,
        )

    @staticmethod
    def _write_to_disk(files: List[GeneratedFile], output_dir: str) -> None:
        for f in files:
            full_path = os.path.join(output_dir, f.path)
            os.makedirs(os.path.dirname(full_path), exist_ok=True)
            with open(full_path, "w", encoding="utf-8") as fh:
                fh.write(f.content)
        print(f"\nProject written to {os.path.abspath(output_dir)}")
